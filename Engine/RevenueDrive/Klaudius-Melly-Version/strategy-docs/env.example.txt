# ============================================================
# Operating country and language
#
# OPERATOR_COUNTRY — full country name (e.g. "Italy", "Spain", "France",
#   "the United Kingdom"). Used in natural-language references throughout
#   the skills and as the default region when /find is invoked without
#   an explicit region argument.
# OPERATOR_COUNTRY_CODE — ISO 3166-1 alpha-2 code matching the country
#   (e.g. IT, ES, FR, GB). Passed to the Google Places API so search results,
#   addresses, and ranking favour the right country.
# OPERATOR_LANGUAGE — full language name (e.g. "Italian", "Spanish",
#   "French", "English"). All outreach, follow-ups, and built websites are
#   written in this language.
# OPERATOR_LANGUAGE_CODE — ISO 639-1 code matching the language (e.g. it,
#   es, fr, en). Used by the Places API for response language and as the
#   value of <html lang="..."> on every built site.
#
# Leave any of these blank to fall back to the United Kingdom / English
# defaults (the original Klaudius operating profile).
# ============================================================

OPERATOR_COUNTRY=the United Kingdom
OPERATOR_COUNTRY_CODE=GB
OPERATOR_LANGUAGE=English
OPERATOR_LANGUAGE_CODE=en

# ============================================================
# Pipeline modes
#
# PIPELINE_MODES — comma-separated lead streams the pipeline targets.
#   - "classic"  businesses with NO website (the default)
#   - "rescue"   ALSO target businesses whose existing website is
#     verifiably dead or bad — dead domain, hosting placeholder, broken
#     HTTPS, doesn't work on phones, visibly ancient. /find gates every
#     rescue candidate through scripts/site-check.js; borderline sites
#     don't qualify.
# Unset or blank = classic only. Set by the init/configure wizard
# ("Also target businesses with a bad or dead existing website?");
# also freely editable here or by asking your agent.

# Klaudius will run /find /gather /build /deploy as normal but stop
# short of /outreach — operator pitches manually or not at all. The
# channel config below is only required when OUTREACH_ENABLED=true.
#
# OUTREACH_CHANNELS is the set of channels you've enabled — a
# comma-separated list drawn from: email, whatsapp, sms.
# OUTREACH_PRIORITY is the source of truth for sending: /outreach walks
# it in order per client and picks the first channel that's viable for
# that client AND succeeds (e.g. "email,whatsapp,sms"). Channels you
# didn't enable never appear in it.
#
# SIGNATURE appears at the bottom of every outreach message

# ============================================================
# Deployment (where the deploy skill ships generated client sites)
#
# DEPLOY_PROVIDER selects the host: "vercel" (default), "cloudflare",
# "netlify", or "selfhost" (your own server over SSH). Set it once; only
# the matching credentials below are read at deploy time. The sites
# Klaudius builds are static Next.js exports, so any of these hosts serve
# them with no adapter — pick whichever you have an account with. To
# switch later, change this line and fill in the other host's credentials
# (or just run `npx klaudius configure`).
#
# ── If DEPLOY_PROVIDER=vercel ──
# VERCEL_TOKEN is a personal API token used by the deploy skill so it can
# run `vercel link / pull / build / deploy` non-interactively. Generate
# one at https://vercel.com/account/tokens (recommended: no expiration).
#
# VERCEL_SCOPE is OPTIONAL. Older single-scope Vercel accounts can leave
# this blank — `vercel link` picks the only scope automatically. Post-2024
# Vercel signups auto-create a team scope (slug like `alice-1234s-projects`)
# rather than using a personal scope, so `vercel link` errors about scope
# without --scope. Set this to your team slug as shown by
# `npx vercel teams ls --token=YOUR_TOKEN`.
#
# ── If DEPLOY_PROVIDER=cloudflare ──
# CLOUDFLARE_API_TOKEN is used by `wrangler pages deploy`. Create it at
# https://dash.cloudflare.com/profile/api-tokens as a Custom Token with
# exactly this permission: Account → Cloudflare Pages → Edit (an
# under-scoped token is the #1 cause of a 403 at deploy time).
# CLOUDFLARE_ACCOUNT_ID is the 32-char hex string from the dashboard URL
# (dash.cloudflare.com/<ACCOUNT_ID>). Required — without it wrangler shows
# an interactive account picker that would stall an unattended run.
#
# ── If DEPLOY_PROVIDER=netlify ──
# NETLIFY_AUTH_TOKEN is a personal access token used by the Netlify CLI so
# the deploy skill can run `netlify sites:create` / `netlify deploy`
# non-interactively. Create one at
# https://app.netlify.com/user/applications#personal-access-tokens
# (recommended: no expiration). A personal access token carries full
# account access, so no separate account ID is needed (unlike Cloudflare).
#
# NETLIFY_ACCOUNT_SLUG is OPTIONAL. Leave it blank if your account has a
# single team — the deploy skill auto-detects it from the token. If you're
# in several Netlify teams, set this to the team slug that should own your
# client sites (Team settings → General → Team details), so new sites land
# in the right team instead of whichever one the API returns first.
#
# ── If DEPLOY_PROVIDER=selfhost ──
# The deploy skill builds each site locally and rsyncs the static export
# to your own server over SSH. Auth is SSH keys only (your key must
# already be authorized on the server — non-standard ports and identity
# files go in ~/.ssh/config, never here).
#
# SELFHOST_DEPLOY_TARGET is the rsync destination base, e.g.
# deploy@203.0.113.7:/var/www/klaudius-sites — each client site uploads
# to <target>/<slug>/.
#
# SELFHOST_URL_TEMPLATE is the public URL pattern with a {slug}
# placeholder, e.g. https://{slug}.sites.example.com. The {slug} must be
# in the HOSTNAME, not the path — static exports use root-absolute asset
# links, so each site has to be served from the root of its own hostname
# (wildcard DNS + a wildcard vhost on your server makes this zero work
# per site).
# ============================================================

DEPLOY_PROVIDER=vercel

VERCEL_TOKEN=your-vercel-api-token
VERCEL_SCOPE=
